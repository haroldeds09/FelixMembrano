local function addPressEffect(btn, targetFrame, getNormalColor)
	targetFrame = targetFrame or btn
	btn.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			TweenService:Create(targetFrame, C.tweenFast, {BackgroundColor3 = C.pressed}):Play()
		end
	end)
	btn.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			local normColor = (type(getNormalColor) == "function" and getNormalColor()) or getNormalColor or C.bg2
			TweenService:Create(targetFrame, C.tweenFast, {BackgroundColor3 = normColor}):Play()
		end
	end)
end

local function fireConnections(signal, ...)
	if typeof(firesignal) == "function" then
		pcall(firesignal, signal, ...)
		return true
	end
	if typeof(getconnections) ~= "function" then return false end
	local ok, connections = pcall(getconnections, signal)
	if not ok or type(connections) ~= "table" then return false end
	local args = table.pack(...)
	local fired = false
	for _, connection in ipairs(connections) do
		local fireOk = pcall(function()
			if connection.Enabled ~= false then
				if args.n > 0 then
					connection:Fire(table.unpack(args, 1, args.n))
				else
					connection:Fire(true)
				end
			end
		end)
		fired = fired or fireOk
	end
	return fired
end

local function getCodeTotalParts(txt)
	if not txt or txt == "" then return 1 end
	local clean = txt:gsub("<[^>]+>", "")
	local parts = {}
	for word in clean:gmatch("%S+") do
		table.insert(parts, word)
	end
	return #parts > 0 and #parts or 1
end

local pendingCodePrefix = false
local pendingCodePrefixTime = 0

-- Auto Start también entiende anuncios dictados por partes:
-- "USE" -> "THIS CODE" -> "FREEPORKS27"
-- o "CODE..." / "CODE:" antes del código.
local pendingSpokenCodePrefix = false
local pendingSpokenCodePrefixTime = 0

local function cleanDetectedCode(code)
	code = tostring(code or "")
	code = code:gsub("<[^>]+>", "")
	code = code:gsub("^[%s%p]+", ""):gsub("[%s%p]+$", "")
	return code
end

local function extractCodeAnnouncement(text)
	if not text or text == "" then return nil end
	local clean = tostring(text):gsub("<[^>]+>", "")
	clean = clean:gsub("^%s+", ""):gsub("%s+$", "")
	local lower = clean:lower()

	-- "use code" puede llegar solo o con el código al lado.
	-- "use code for ..." NO contiene un código: "for" es parte de la frase.
	local useA, useB = lower:find("use%s+code")
	if useA then
		local after = clean:sub(useB + 1)
		after = after:gsub("^%s+", ""):gsub("^[:%-–—]+%s*", "")
		after = after:gsub("[%s%.!%?…]+$", "")
		after = after:gsub("^%s+", ""):gsub("%s+$", "")

		local afterLower = after:lower()
		if afterLower == "" then
			pendingCodePrefix = true
			pendingCodePrefixTime = tick()
			return nil
		end

		-- Nunca canjear "for", "for a meowl", etc.
		if afterLower:match("^for%s") or afterLower == "for" then
			pendingCodePrefix = false
			return nil
		end

		-- Cuando el código está directamente al lado: "use code FREEPORKS27".
		-- Aceptamos un solo token alfanumérico/guiones/underscores, no una frase.
		local directCode = after:match("^([%w_%-]+)$")
		if directCode and directCode ~= "" then
			pendingCodePrefix = false
			return cleanDetectedCode(directCode)
		end

		pendingCodePrefix = false
		return nil
	end

	-- "The code is" / "code is" puede traer el código en el mismo anuncio.
	local patterns = {
		"the%s+code%s+is",
		"code%s+is"
	}

	for _, pattern in ipairs(patterns) do
		local a, b = lower:find(pattern)
		if a then
			local after = clean:sub(b + 1)
			after = after:gsub("^%s+", ""):gsub("^[:%-–—]+%s*", "")
			after = after:gsub("[%s%.!%?…]+$", "")
			after = after:gsub("^%s+", ""):gsub("%s+$", "")

			local normalizedAfter = after:lower():gsub("[%p%s…]+", "")
			local fakeOnly = {
				[""] = true,
				["code"] = true,
				["usecode"] = true,
				["thecodeis"] = true
			}

			if normalizedAfter ~= "" and not fakeOnly[normalizedAfter] then
				local code = cleanDetectedCode(after)
				if code ~= "" then
					pendingCodePrefix = false
					return code
				end
			end

			pendingCodePrefix = true
			pendingCodePrefixTime = tick()
			return nil
		end
	end

	-- Formato: "drgoni256 is The code"
	local codeBefore = clean:match('^["“”]?([^"“”]+)["“”]?%s+[Ii][Ss]%s+[Tt][Hh][Ee]%s+[Cc][Oo][Dd][Ee]%s*[%.!%?…]*$')
	if codeBefore then
		codeBefore = cleanDetectedCode(codeBefore)
		if codeBefore ~= "" then
			pendingCodePrefix = false
			return codeBefore
		end
	end

	-- Si "The code is" llegó separado, el siguiente anuncio es el código.
	if pendingCodePrefix and tick() - pendingCodePrefixTime <= 2 then
		pendingCodePrefix = false
		local nextCode = cleanDetectedCode(clean)
		local nextNorm = nextCode:lower():gsub("[%p%s…]+", "")
		if nextCode ~= "" and nextNorm ~= "code" and nextNorm ~= "usecode" and nextNorm ~= "thecodeis" and nextNorm ~= "for" then
			return nextCode
		end
	end

	pendingCodePrefix = false
	return nil
end

local function finishSubmit()
	-- Submit solo detiene el ciclo actual de Start.
	-- Auto Start es una configuración independiente y permanece ON.
	stopStartAfterSubmit()
end

local function dispatch(text)
	if not text or text == "" then return end
	if tick() < suppressDispatchUntil then return end

	local cleanText = tostring(text):gsub("<[^>]+>", "")
	local lowerText = cleanText:lower()
	local trimmedText = cleanText:gsub("^%s+", ""):gsub("%s+$", "")
	local trimmedLower = trimmedText:lower()

	-- Auto Start reconoce anuncios de código DICTADOS/PARTIDOS.
	-- No dependemos de una frase exacta: mantenemos una pequeña ventana
	-- de contexto para entender frases como:
	-- USE -> THE CODE FOR -> USE THIS CODE -> USE THIS -> FREEPORKS27
	-- USE THIS CODE -> CODE... -> FREEPORKS27
	local spokenCodeLead = false
	local spokenLeadOnly = false
	local spokenResolvedCode = nil

	local function normalizeSpokenLead(txt)
		local t = tostring(txt or ""):gsub("<[^>]+>", "")
		t = t:gsub("^%s+", ""):gsub("%s+$", ""):lower()
		-- La puntuación final es ruido de voz/transcripción.
		t = t:gsub("[%s%.%!%?…:;%-]+$", "")
		t = t:gsub("%s+", " ")
		return t
	end

	local function isSpokenCodeLead(txt)
		local t = normalizeSpokenLead(txt)
		return
			t == "use"
			or t == "use this"
			or t == "this code"
			or t == "use this code"
			or t == "code"
			or t == "code for"
			or t == "use code"
			or t == "use code for"
			or t == "the code"
			or t == "the code for"
			or t == "use the code"
			or t == "use the code for"
	end

	-- También aceptamos frases que claramente terminan en un marcador
	-- de código, aunque tengan palabras intermedias.
	local function looksLikeSpokenCodeLead(txt)
		local t = normalizeSpokenLead(txt)
		if isSpokenCodeLead(t) then return true end
		if t:match("^use%s+the%s+code%s*$") then return true end
		if t:match("^use%s+the%s+code%s+for%s*$") then return true end
		if t:match("^use%s+code%s*$") then return true end
		if t:match("^use%s+code%s+for%s*$") then return true end
		if t:match("^use%s+this%s+code%s*$") then return true end
		if t:match("^this%s+code%s*$") then return true end
		if t == "code" then return true end
		return false
	end

	if looksLikeSpokenCodeLead(trimmedText) then
		spokenCodeLead = true
		spokenLeadOnly = true
	end

	if pendingSpokenCodePrefix then
		if tick() - pendingSpokenCodePrefixTime <= 6 then
			if looksLikeSpokenCodeLead(trimmedText) then
				-- Sigue llegando la introducción. Renovamos la ventana.
				pendingSpokenCodePrefixTime = tick()
				spokenCodeLead = true
				spokenLeadOnly = true
			else
				-- Un código real normalmente llega como un solo token.
				-- Frases como "a meowl" no se aceptan como código.
				local possibleSpokenCode = trimmedText:match("^([%w_%-]+)$")
				local possibleLower = possibleSpokenCode and possibleSpokenCode:lower() or ""

				if possibleSpokenCode
					and possibleLower ~= "use"
					and possibleLower ~= "this"
					and possibleLower ~= "code"
					and possibleLower ~= "for"
					and possibleLower ~= "the"
				then
					pendingSpokenCodePrefix = false
					local announcedFromSpokenPrefix = cleanDetectedCode(possibleSpokenCode)

					-- Si el usuario acaba de dictar un prefijo inequívoco,
					-- Auto Start se enciende automáticamente y Start también.
					autoStartEnabled = true
					if autoStartBtn then
						autoStartBtn.Text = coloredToggleText("Auto Start", true)
					end

					if not isScanning then
						isScanning = true
						if startBtn then startBtn.Text = coloredToggleText("Start", true) end
						if statusDot then statusDot.BackgroundColor3 = C.statusGreen end
						collectBuf = {}
						collectProgress = 0
						lastCollectedText = ""
						lastDetectedCodeFragment = ""
					end

					spokenResolvedCode = announcedFromSpokenPrefix
					cleanText = announcedFromSpokenPrefix
					lowerText = cleanText:lower()
					trimmedText = cleanText
					trimmedLower = lowerText
					spokenCodeLead = false
					spokenLeadOnly = false
				else
					-- Si no parece un token de código todavía, seguimos esperando.
					pendingSpokenCodePrefixTime = tick()
				end
			end
		else
			pendingSpokenCodePrefix = false
		end
	end

	if spokenCodeLead then
		-- Un prefijo hablado activa Auto Start aunque el botón estuviera OFF.
		autoStartEnabled = true
		if autoStartBtn then
			autoStartBtn.Text = coloredToggleText("Auto Start", true)
		end
		pendingSpokenCodePrefix = true
		pendingSpokenCodePrefixTime = tick()
	end

	-- Cualquier prefijo detectado activa Start, pero nunca se canjea como código.
	local startAnnouncement =
		lowerText:find("use%s+code") ~= nil
		or lowerText:find("the%s+code%s+is") ~= nil
		or lowerText:find("code%s+is") ~= nil
		or lowerText:find("use%s+the%s+code") ~= nil
		or spokenCodeLead

	if startAnnouncement then
		autoStartEnabled = true
		if autoStartBtn then
			autoStartBtn.Text = coloredToggleText("Auto Start", true)
		end
		if not isScanning then
			isScanning = true
			if startBtn then startBtn.Text = coloredToggleText("Start", true) end
			if statusDot then statusDot.BackgroundColor3 = C.statusGreen end
			collectBuf = {}
			collectProgress = 0
			lastCollectedText = ""
			lastDetectedCodeFragment = ""
		end
	end

	-- Los prefijos hablados nunca pasan al flujo de redeem.
	if spokenLeadOnly then
		return
	end

	local announcedCode = spokenResolvedCode or extractCodeAnnouncement(cleanText)

	-- "USE", "USE THIS CODE" y "CODE.../CODE:" son solo disparadores
	-- de Auto Start; nunca deben entrar al canje como si fueran códigos.
	if autoStartEnabled and spokenLeadOnly and not announcedCode then
		return
	end

	-- "Use code" es solamente el disparador de Start.
	-- No debe terminar entrando al flujo de redeem.
	if lowerText:find("use%s+code") and not announcedCode then
		return
	end

	if announcedCode and announcedCode ~= "" then
		-- Un código anunciado con prefijo completo (por ejemplo
		-- "use code FREEPORKS27" o "The code is FREEPORKS27")
		-- conserva el comportamiento automático: si Start estaba OFF,
		-- se enciende y el código entra al flujo de Submit.
		if not isScanning then
			isScanning = true
			if startBtn then startBtn.Text = coloredToggleText("Start", true) end
			if statusDot then statusDot.BackgroundColor3 = C.statusGreen end
			collectBuf = {}
			collectProgress = 0
			lastCollectedText = ""
			lastDetectedCodeFragment = ""
		end

		-- Un código anunciado completo cuenta como UNA entrega.
		-- Solo se canjea al alcanzar el número elegido en Submit Code After.
		local submitTarget = (captureCount and captureCount > 0) and captureCount or 1
		collectProgress = math.min(collectProgress + 1, submitTarget)
		table.insert(collectBuf, announcedCode)
		local currentCombined = table.concat(collectBuf)

		lastDetectedCodeFragment = announcedCode
		lastSubmitAmount = collectProgress
		logStatus(currentCombined .. " | " .. tostring(collectProgress) .. "/" .. tostring(submitTarget), true)

		if combineEnabled then
			-- Spam & Reedem: canjea el acumulado en cada fragmento.
			redeemCode(currentCombined)
			if collectProgress >= submitTarget then
				finishSubmit()
			end
		elseif collectProgress >= submitTarget then
			-- Normal: canjea únicamente al llegar al número configurado.
			redeemCode(currentCombined)
			finishSubmit()
			collectBuf = {}
			collectProgress = 0
			lastCollectedText = ""
			activeDynamicLabel = nil
		end
		return
	end

	if not isScanning then return end

	-- Los riddles reales siguen pasando al Auto Riddle / solver.
	-- No descartamos respuestas como SAMMY/PORK/24 aquí: si el texto
	-- no es una pregunta, se trata como fragmento del announcement.
	if isRiddleText(text) and tostring(text):find("%?") then
		return
	end

	if #collectBuf == 0 and not activeDynamicLabel then
		clearLogUI()
	end

	local submitTarget = (captureCount and captureCount > 0) and captureCount or 1

	-- Cada anuncio recibido es un fragmento independiente.
	-- Esto permite secuencias como:
	-- SAMMY
	-- PORK
	-- 24
	-- y solo hace redeem al llegar a 3/3 (o al número configurado).
	table.insert(collectBuf, cleanDetectedCode(text))
	collectProgress = math.min(collectProgress + 1, submitTarget)
	lastSubmitAmount = collectProgress

	local currentCombined = table.concat(collectBuf)
	lastCollectedText = text
	lastDetectedCodeFragment = cleanDetectedCode(text)

	logStatus(currentCombined .. " | " .. tostring(collectProgress) .. "/" .. tostring(submitTarget), true)

	if combineEnabled then
		-- Spam & Reedem: canjea cada acumulación sin esperar al límite.
		redeemCode(currentCombined)
		if collectProgress >= submitTarget then
			finishSubmit()
		end
	elseif collectProgress >= submitTarget then
		-- Normal: espera exactamente al Submit Code After seleccionado.
		redeemCode(currentCombined)
		finishSubmit()
		collectBuf = {}
		collectProgress = 0
		lastCollectedText = ""
		activeDynamicLabel = nil
	end
end

local function watchObject(obj)
	if seen[obj] or isOwnedByUs(obj) then return end
	seen[obj] = true
	if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
		obj:GetPropertyChangedSignal("Text"):Connect(function()
			if not isOwnedByUs(obj) and obj.Text ~= "" then
				dispatch(obj.Text)
			end
		end)
	end
end

local function watchTree(obj)
	if isOwnedByUs(obj) then return end
	watchObject(obj)
	for _, child in ipairs(obj:GetDescendants()) do
		watchObject(child)
	end
	obj.DescendantAdded:Connect(function(child)
		if isOwnedByUs(child) then return end
		local isNew = not seen[child]
		watchObject(child)
		if isNew then
			local t = getText(child)
			if t and t ~= "" then
				dispatch(t)
			end
		end
	end)
end

local function hookContainers()
	local names = { "TopNotification" }
	for _, name in ipairs(names) do
		local c = PlayerGui:FindFirstChild(name)
		if c then watchTree(c) end
	end
	PlayerGui.ChildAdded:Connect(function(child)
		for _, name in ipairs(names) do
			if child.Name == name then
				watchTree(child)
			end
		end
	end)
end

localExactAnswer = function(text)
	if not text or text == "" then return nil end

	local function normalize(value)
		value = stripRich(tostring(value or "")):lower()
		value = value:gsub("[%c]", " ")
		value = value:gsub("[%p]", " ")
		value = value:gsub("%s+", " ")
		return trim(value)
	end

	-- Conservamos comas y signos + antes de normalizar, porque pueden
	-- separar varias pistas dentro de UN SOLO mensaje.
	local rawText = stripRich(tostring(text or "")):lower()
	rawText = rawText:gsub("[%c]", " ")
	rawText = rawText:gsub("%s+", " ")
	rawText = trim(rawText)

	local clean = normalize(text)
	if clean == "" then return nil end

	-- ============================================================
	-- PRIORIDAD SEMÁNTICA
	-- Estas pistas deben ganar antes que coincidencias generales
	-- como "best brainrot" o "first OG".
	-- ============================================================
	local hasCraftMachine = clean:find("craft machine", 1, true) ~= nil
	local hasBackrooms = clean:find("backrooms", 1, true) ~= nil
	local asksSecond = clean:find("second", 1, true) ~= nil
		or clean:find("2nd", 1, true) ~= nil
	local asksBest = clean:find("best", 1, true) ~= nil
		or clean:find("number one", 1, true) ~= nil
		or clean:find("number 1", 1, true) ~= nil
		or clean:find("top", 1, true) ~= nil
		or clean:find("most op", 1, true) ~= nil
		or clean:find("greatest", 1, true) ~= nil
		or clean:find("strongest", 1, true) ~= nil

	if hasCraftMachine then
		if asksSecond then
			return "LACRAFTMACHINE"
		end
		if asksBest then
			return "LOSDRAGONS"
		end
	end

	if hasBackrooms and asksBest then
		return "RUBRIKIKO"
	end

	-- Respuestas semánticas directas adicionales.
	if clean:find("favorite artist", 1, true) or clean:find("favourite artist", 1, true) then
		return "DRAKE"
	end
	if clean:find("month", 1, true) and (clean:find("sab came out", 1, true)
		or clean:find("sab released", 1, true) or clean:find("game came out", 1, true)
		or clean:find("game released", 1, true)) then
		return "MAY"
	end
	if clean:find("year", 1, true) and (clean:find("sab came out", 1, true)
		or clean:find("sab released", 1, true) or clean:find("game came out", 1, true)
		or clean:find("game released", 1, true)) then
		return "2025"
	end
	if clean:find("fought", 1, true) and clean:find("admin war", 1, true) then
		return "JANDEL"
	end
	if clean:find("represents good", 1, true) or clean:find("represents the good", 1, true) then
		return "DIVINE"
	end
	if clean:find("represents evil", 1, true) or clean:find("represents bad", 1, true)
		or clean:find("represents the evil", 1, true) then
		return "CURSED"
	end

	local function solvePart(part)
		part = normalize(part)
		if part == "" then return nil end

		local bestVal = nil
		local bestLen = 0
		for key, val in pairs(SAB_DB) do
			local k = normalize(key)
			if k ~= "" then
				local pos = part:find(k, 1, true)
				if pos and #k > bestLen then
					bestVal = val
					bestLen = #k
				end
			end
		end
		-- Permitir números literales como una parte del riddle (ej. 67).
		if not bestVal and part:match("^%d+$") then
			return part
		end
		return bestVal
	end

	-- Los riddles compuestos pueden venir separados por + o comas.
	-- No se divide automáticamente por "and", porque puede formar parte
	-- de una pista individual, como "black and white mutation".
	local parts = {}
	local compoundText = rawText:gsub(",", "+")
	if compoundText:find("%+") then
		for part in compoundText:gmatch("[^%+]+") do
			table.insert(parts, part)
		end
	end

	if #parts >= 2 then
		local combined = ""
		local allSolved = true
		for _, part in ipairs(parts) do
			local answer = solvePart(part)
			if not answer then
				allSolved = false
				break
			end
			combined = combined .. tostring(answer):upper():gsub("%s+", "")
		end
		if allSolved and combined ~= "" then
			return combined
		end
	end

	return solvePart(clean)
end

local function queryAI(question, statusCb)
	if not httpRequest then return nil, nil, "no http executor" end
	if not GROQ_API_KEY or GROQ_API_KEY == "" then return nil, nil, "GROQ_API_KEY is empty" end
	if statusCb then statusCb("Calling AI...") end
	local body = HttpService:JSONEncode({
		model = "llama-3.3-70b-versatile",
		messages = {
			{ role = "system", content = AI_SYSTEM },
			{ role = "user", content = "/no_think\n" .. tostring(question) .. "\n\nRules: ONE word, ALL CAPS, NO spaces, NO repetition, NO spelling out numbers. Just the answer." },
		},
		temperature = 0,
		max_tokens = 64,
	})
	local ok, res = pcall(httpRequest, {
		Url = "https://api.groq.com/openai/v1/chat/completions",
		Method = "POST",
		Headers = {
			["Content-Type"] = "application/json",
			["Authorization"] = "Bearer " .. GROQ_API_KEY,
		},
		Body = body,
	})
	if not ok or not res or not res.Body then return nil, nil, "HTTP Error" end
	local pOk, parsed = pcall(HttpService.JSONDecode, HttpService, res.Body)
	if not pOk or not parsed or not parsed.choices or not parsed.choices[1] then return nil, nil, "Parse Error" end
	local answer = (parsed.choices[1].message and parsed.choices[1].message.content) or ""
	answer = trim(answer):upper():gsub("[%s%?%.%,!\"'`\n\r]+", "")
	if answer:find(":") then answer = answer:match(":([^:]+)$") or answer end
	return answer, nil, nil
end

local function solveSingleRiddle(text, statusCb)
	text = trim(tostring(text or ""))
	if text == "" then return nil end

	local a = localExactAnswer(text)
	if a then return a end

	local aiAns = select(1, queryAI(text, statusCb))
	if aiAns and aiAns ~= "" then
		return aiAns
	end
	return nil
end

local function solveRiddle(text, statusCb)
	text = trim(tostring(text or ""))
	if text == "" then return nil, nil, "No answer" end

	-- Algunas notificaciones meten VARIAS preguntas dentro del mismo texto,
	-- por ejemplo: "Pregunta 1? Pregunta 2? Pregunta 3?".
	-- Antes se mandaba todo el bloque a la IA como una sola pregunta,
	-- haciendo que la respuesta pudiera salir pegada al texto original.
	-- Ahora cada pregunta se resuelve por separado y al final se concatenan
	-- SOLO las respuestas, en el mismo orden.
	local questions = {}
	for part in text:gmatch("([^%?]+%?)") do
		part = trim(part)
		if part ~= "" then
			table.insert(questions, part)
		end
	end

	local withoutQuestions = text:gsub("[^%?]+%?", " ")
	withoutQuestions = trim(withoutQuestions)
	if withoutQuestions ~= "" and #questions > 0 then
		table.insert(questions, withoutQuestions)
	elseif #questions == 0 then
		questions = { text }
	end

	if #questions > 1 then
		local combined = ""
		for _, question in ipairs(questions) do
			local answer = solveSingleRiddle(question, statusCb)
			if not answer or answer == "" then
				return nil, nil, "No answer for one of the questions"
			end
			combined = combined .. tostring(answer):upper():gsub("%s+", "")
		end
		return combined, nil, nil
	end

	local answer = solveSingleRiddle(text, statusCb)
	if answer then return answer, nil, nil end
	return nil, nil, "No answer"
end

local RIDDLE_POSITIONS = { Top = true, Bottom = true, Center = true, Middle = true }
local riddleListenConn = nil
local riddleSolving = false
local riddleQueue = {}
local riddleQueued = {}
local riddleRecent = {}
local riddleAnswered = {}
local riddleAccumulatedAnswer = ""

local function normalizeRiddleKey(text)
	text = stripRich(tostring(text or "")):lower()
	text = text:gsub("[%c]", " ")
	text = text:gsub("%s+", " ")
	return trim(text)
end

local function looksLikeRiddleAnnouncement(...)
	local args = table.pack(...)
	if args.n == 0 or typeof(args[1]) ~= "string" then return false end
	for i = 2, args.n do
		local value = args[i]
		if typeof(value) == "string" and (value:find("Sounds%.") or value:find("rbxassetid") or RIDDLE_POSITIONS[value]) then
			return true
		end
	end
	return false
end

local function resolveRiddleNotifyRemote()
	local ok, controller = pcall(function()
		local controllers = ReplicatedStorage:FindFirstChild("Controllers")
		local notification = controllers and controllers:FindFirstChild("NotificationController", true)
		return notification and require(notification) or nil
	end)
	if not ok or type(controller) ~= "table" or typeof(getupvalues) ~= "function" then return nil end
	local valuesOk, values = pcall(getupvalues, controller.Start)
	if not valuesOk or type(values) ~= "table" then return nil end
	for _, value in pairs(values) do
		if typeof(value) == "Instance" and (value:IsA("RemoteEvent") or value:IsA("UnreliableRemoteEvent")) then
			return value
		end
	end
	return nil
end

local function processAutoRiddle(...)
	if not autoRiddleEnabled then return end

	local text = stripRich(tostring((...) or ""))
	text = trim(text)
	if text == "" then return end

	-- El mismo anuncio puede llegar varias veces desde el sistema de
	-- notificaciones. No lo volvemos a resolver ni a añadir.
	local key = normalizeRiddleKey(text)
	local now = tick()
	if riddleAnswered[key] then
		return
	end
	if riddleRecent[key] and (now - riddleRecent[key]) < 3 then
		return
	end
	riddleRecent[key] = now

	if riddleQueued[key] then return end
	riddleQueued[key] = true
	table.insert(riddleQueue, { text = text, key = key })

	if riddleSolving then return end
	riddleSolving = true

	task.spawn(function()
		while autoRiddleEnabled and #riddleQueue > 0 do
			local item = table.remove(riddleQueue, 1)
			riddleQueued[item.key] = nil

			-- Resolver el riddle sin crear una línea nueva por cada pregunta.
			-- Igual que Start, mantenemos UNA sola línea dinámica con el
			-- texto acumulado y la vamos reemplazando/actualizando.
			local answer = solveRiddle(item.text, function(msg)
				-- No añadir mensajes intermedios al log.
			end)

			if answer and answer ~= "" then
				riddleAnswered[item.key] = true
				answer = tostring(answer):upper():gsub("%s+", "")

				-- Si el mismo riddle vuelve después del periodo de deduplicación,
				-- no lo acumulamos dos veces.
				if riddleAccumulatedAnswer == "" then
					riddleAccumulatedAnswer = answer
				else
					riddleAccumulatedAnswer = riddleAccumulatedAnswer .. answer
				end

				-- UNA SOLA línea: se actualiza en lugar de crear otra debajo.
				logStatus(riddleAccumulatedAnswer, true)
				submitRiddleAnswer(riddleAccumulatedAnswer)
			else
				-- No crear una línea adicional si no se pudo resolver.
			end

			task.wait(0.05)
		end

		riddleSolving = false
	end)
end

local function connectAutoRiddleListener()
	if riddleListenConn then return true end
	local notifyRemote = resolveRiddleNotifyRemote()
	if not notifyRemote then return false end
	local ok, conn = pcall(function()
		return notifyRemote.OnClientEvent:Connect(function(...)
			if looksLikeRiddleAnnouncement(...) then processAutoRiddle(...) end
		end)
	end)
	if ok and conn then
		riddleListenConn = conn
		return true
	end
	return false
end

local function setAutoRiddleEnabled(state, silent)
	autoRiddleEnabled = state == true
	if autoRiddleEnabled then
		if connectAutoRiddleListener() then
			-- Sin log al activar Auto riddle
		else
			-- Sin log al activar Auto riddle
		end
	else
		if riddleListenConn then
			riddleListenConn:Disconnect()
			riddleListenConn = nil
		end
		riddleQueue = {}
		riddleQueued = {}
		riddleRecent = {}
		riddleAnswered = {}
		riddleAccumulatedAnswer = ""
		riddleSolving = false
		-- Sin log al desactivar Auto riddle
	end
end

local function checkSpawn(obj)
	if not obj:IsA("TextLabel") then return end
	local txt = obj.Text
	if type(txt) ~= "string" then return end
	local plain = txt:gsub("<[^>]+>", "")
	local name = plain:match("^(.-)%s+spawned!%s*$")
	if name and name ~= "" then
		-- Code/Spawned detectado: detener Start automáticamente.
		isScanning = false
		collectBuf = {}
		collectProgress = 0
		lastCollectedText = ""
		if startBtn then
			startBtn.Text = coloredToggleText("Start", false)
		end
		if statusDot then
			statusDot.BackgroundColor3 = C.statusRed
		end

		-- El spawn reemplaza por completo el acumulado anterior.
		-- Durante un instante ignoramos avisos atrasados que puedan llegar
		-- de la misma notificación para que no aparezcan debajo del spawn.
		suppressDispatchUntil = tick() + 0.75
		clearLogUI()
		local completedSubmit = (captureCount and captureCount > 0) and captureCount or 1
		logStatus(name .. " | " .. tostring(completedSubmit) .. "/" .. tostring(completedSubmit), false)
		sendWebhook(name)
	end
end

-- Auto Buy trabaja sobre ProximityPrompts relacionados con Buy/Purchase/Compra.
local buyPrompts = {}

local function isBuyPrompt(prompt)
	if not prompt:IsA("ProximityPrompt") then
		return false
	end

	local action = string.lower(tostring(prompt.ActionText or ""))
	local object = string.lower(tostring(prompt.ObjectText or ""))

	local names = {}
	local current = prompt
	for _ = 1, 6 do
		if not current then break end
		table.insert(names, string.lower(tostring(current.Name or "")))
		current = current.Parent
	end

	local hierarchy = table.concat(names, " ")

	return action:find("buy", 1, true)
		or action:find("purchase", 1, true)
		or action:find("compr", 1, true)
		or object:find("buy", 1, true)
		or object:find("purchase", 1, true)
		or object:find("compr", 1, true)
		or hierarchy:find("buy", 1, true)
		or hierarchy:find("purchase", 1, true)
		or hierarchy:find("compr", 1, true)
end

local function registerBuyPrompt(prompt)
	if isBuyPrompt(prompt) then
		buyPrompts[prompt] = true
	end
end

	if instance:IsA("ProximityPrompt") then
		registerBuyPrompt(instance)
	end
end

workspace.DescendantAdded:Connect(function(instance)
	if instance:IsA("ProximityPrompt") then
		registerBuyPrompt(instance)
	end
end)

workspace.DescendantRemoving:Connect(function(instance)
	if instance:IsA("ProximityPrompt") then
		buyPrompts[instance] = nil
	end
end)

		if autoBuyActive then
			for prompt in pairs(buyPrompts) do
				if prompt.Parent and prompt.Enabled and isBuyPrompt(prompt) then
					prompt.HoldDuration = 0

							break
						end
						prompt:InputHoldBegin()
						prompt:InputHoldEnd()
					end
				end
			end
		end
	end
end)

local function makeKeybind(text, xOffset)
	local btn = Instance.new("TextButton")
	btn.Size = UDim2.new(0, 66, 0, 42)
	btn.AnchorPoint = Vector2.new(1, 0)
	btn.Position = UDim2.new(1, xOffset, 0, 80)
	btn.BackgroundColor3 = C.bg2
	btn.BorderSizePixel = 0
	btn.Font = Enum.Font.GothamBold
	btn.TextSize = 12
	btn.TextColor3 = C.textPri
	btn.Text = text
	btn.Visible = false
	btn.AutoButtonColor = false
	btn.Parent = keybindGui
	corner(btn, UDim.new(0, 8))
	stroke(btn, C.border, 1)

	-- Each floating keybind can be dragged independently.
	local draggingKeybind = false
	local dragStartKeybind = Vector3.zero
	local startPosKeybind = btn.Position
	local activeKeybindInput = nil

	btn.InputBegan:Connect(function(input)
		if isLocked then return end
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			draggingKeybind = true
			dragStartKeybind = input.Position
			startPosKeybind = btn.Position
			activeKeybindInput = input

			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					draggingKeybind = false
					activeKeybindInput = nil
				end
			end)
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if not draggingKeybind or isLocked then return end
		if input == activeKeybindInput
			or input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch then
			local delta = input.Position - dragStartKeybind
			btn.Position = UDim2.new(
				startPosKeybind.X.Scale,
				startPosKeybind.X.Offset + delta.X,
				startPosKeybind.Y.Scale,
				startPosKeybind.Y.Offset + delta.Y
			)
		end
	end)

	return btn
end

local function updatePageRows()
	if not hasCustomResize then return end

	local totalH = currentResizeHeight()
	local contentHeight = math.max(0, totalH - TITLE_H - TABBAR_H)
	local innerHeight = math.max(0, contentHeight - PAGE_PAD * 2)

	-- Main: 6 filas siempre visibles, repartiendo el espacio disponible.
	local mainRowH = math.max(1, math.floor((innerHeight - ROW_GAP * 5) / 6))
	for _, row in ipairs(mainRows) do
		row.Size = UDim2.new(1, 0, 0, mainRowH)
	end

	-- Help: 3 botones + el espacio reservado por Key Helper.
	local helpRowH = math.max(1, math.floor((innerHeight - ROW_GAP * 2 - 30) / 3))
	for _, row in ipairs(helpRows) do
		row.Size = UDim2.new(1, 0, 0, helpRowH)
	end
end

local function minimumHeightForTab(tab)
	if tab == 1 then
		-- Altura mínima que permite mostrar las 5 filas completas dentro del menú.
		return TITLE_H + TABBAR_H + PAGE_PAD * 2 + ROW_GAP * 5 + (20 * 6)
	elseif tab == 2 then
		return TITLE_H + TABBAR_H + PAGE_PAD * 2 + ROW_GAP * 2 + (20 * 3) + 30
	end
	-- Logs no tiene la lista de botones de Main/Help; conserva una altura compacta.
	return TITLE_H + TABBAR_H + 60
end

local function getLayoutSize()
	if isMinimized then
		return 310, MIN_H
	end
	if not hasCustomResize then
		return resizeWidth, currentTabHeight()
	end
	return resizeWidth, currentResizeHeight()
end

local function applyLayout(instant)
	local targetW, targetH = getLayoutSize()
	updatePageRows()
	contentFrame.Visible = not isMinimized
	tabBar.Visible = not isMinimized
	resizeHandle.Visible = not isMinimized
	minBtn.Text = isMinimized and "+" or "-"
	if instant then
		mainFrame.Size = UDim2.new(0, targetW, 0, targetH)
	else
		TweenService:Create(mainFrame, C.tweenMed, { Size = UDim2.new(0, targetW, 0, targetH) }):Play()
	end
end

local function setActiveTab(idx)
	currentTab = idx
	page1.Visible = (idx == 1)
	page2.Visible = (idx == 2)
	page3.Visible = (idx == 3)

	tabMainBtn.BackgroundColor3 = idx == 1 and C.accent or C.bg2
	tabHelpBtn.BackgroundColor3 = idx == 2 and C.accent or C.bg2
	tabLogBtn.BackgroundColor3 = idx == 3 and C.accent or C.bg2

	tabMainBtn.TextColor3 = idx == 1 and C.activeTabText or C.textMuted
	tabHelpBtn.TextColor3 = idx == 2 and C.activeTabText or C.textMuted
	tabLogBtn.TextColor3 = idx == 3 and C.activeTabText or C.textMuted

	-- Al cambiar de apartado, el alto se adapta automáticamente al apartado.
	-- El ancho NO cambia.
	if hasCustomResize and not isMinimized then
		local savedH = resizeHeights[idx]
		if not savedH then
			savedH = currentTabHeight()
		end
		local minH = minimumHeightForTab(idx)
		resizeHeights[idx] = math.max(savedH, minH)
		updatePageRows()
	end
end

addPressEffect(tabLogBtn, tabLogBtn, function() return currentTab == 3 and C.accent or C.bg2 end)

resizeHandle.InputBegan:Connect(function(input)
	if isLocked or isMinimized then return end
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		resizing = true
		resizeStart = input.Position
		resizeStartWidth = resizeWidth
		resizeStartHeight = currentResizeHeight()
		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				resizing = false
			end
		end)
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if not resizing or isLocked or isMinimized then return end
	if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
		local delta = input.Position - resizeStart

		-- El ancho se mueve con X y la altura con Y, exactamente desde la esquina.
		-- Hacia abajo/derecha = más grande; hacia arriba/izquierda = más pequeño.
		local newWidth = math.clamp(resizeStartWidth + delta.X, 250, 600)
		local minResizeHeight = minimumHeightForTab(currentTab)
		local newHeight = math.clamp(resizeStartHeight + delta.Y, minResizeHeight, 560)

		resizeWidth = newWidth
		resizeHeights[currentTab] = newHeight
		hasCustomResize = true

		mainFrame.Size = UDim2.new(0, newWidth, 0, newHeight)
		updatePageRows()
	end
end)

	-- Sin log al bloquear/desbloquear

local dragging = false
local dragStart = Vector3.zero
local startPos = UDim2.new(0, 0, 0, 0)
local activeInput = nil

titleBar.InputBegan:Connect(function(input)
	if isLocked then return end
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = true
		dragStart = input.Position
		startPos = mainFrame.Position
		activeInput = input
		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				dragging = false
				activeInput = nil
				-- No guardar posición: la próxima ejecución siempre inicia centrada.
			end
		end)
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if not dragging or isLocked then return end
	if input == activeInput or input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
		local delta = input.Position - dragStart
		mainFrame.Position = UDim2.new(
			startPos.X.Scale,
			startPos.X.Offset + delta.X,
			startPos.Y.Scale,
			startPos.Y.Offset + delta.Y
		)
	end
end)
